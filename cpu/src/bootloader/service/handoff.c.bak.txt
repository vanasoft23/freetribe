__attribute__((noreturn))
void handoff_factory_firmware(void) {

    // Disable interrupts
    CPUirqd();
    CPUfiqd();
    IntGlobalDisable();
    // HWREG(SOC_SPI_1_REGS + SPI_SPIPC(0)) = 0x01010e00;
    // HWREG(SOC_SPI_1_REGS + SPI_SPIPC(1)) = 0x00000003;

    // Uninitialize peripherals used by bootloader that
    // firmware expects not to be initialized.
    per_spi_unregister_interrupts(0); // LCD
    boot_usb_terminate();
    per_uart_terminate(0); // MCU UART
    dev_sdcard_terminate();

    // Match GPIO
    GPIOBankOutputDataSet(SOC_GPIO_0_REGS, 0, 0x0);
    GPIOBankDirModeSet(SOC_GPIO_0_REGS, 0, 0xffffffff);

    // Match pinmux. Factory flash driver bitbangs SPI CS pins
    HWREG(SOC_SYSCFG_0_REGS + SYSCFG0_PINMUX(5)) = 0x81118188;

    // If GDB uploaded an image, make it physical,
    // and discard dirty bootloader state.
    CP15DCacheCleanFlush();
    _drain_write_buffer();

    // Zero remaining 62 MiB DDR, just in case.
    // Note that we use uncached DDR alias, that's why we start at 0xC4200000.
    u32 *ptr = (u32 *)0xC4200000;
    u32 *end = (u32 *)0xC8000000;
    while (ptr < end)
        *ptr++ = 0;

    CP15ICacheFlush();
    _drain_write_buffer();

    // Point vector table to DDR
    CP15HighVectorDisable();
    HWREG(SOC_AINTC_0_REGS + AINTC_VBR) = 0xC0000000;

    // Jump to entry point
    // @TODO: MAYBE DONT SWITCH SVC MODE HERE, BUT EXIT ISR CONTEXT
    asm volatile(
        // "msr   cpsr_c, #0xD3   \n\t" // SVC mode, IRQ/FIQ masked, ARM state
        "ldr pc, =0xC0000000"
        : : : "memory"
    );
    __builtin_unreachable();

}

__attribute__((noreturn))
void handoff_freetribe_kernel(void) {

    // Disable interrupts
    CPUirqd();
    CPUfiqd();
    IntGlobalDisable();

    // Uninitialize peripherals used by bootloader that
    // firmware expects not to be initialized.
    per_spi_unregister_interrupts(0); // LCD
    boot_usb_terminate();
    per_uart_terminate(0); // MCU UART
    dev_sdcard_terminate();

    // Match GPIO
    // GPIOBankOutputDataSet(SOC_GPIO_0_REGS, 0, 0x0);
    // GPIOBankDirModeSet(SOC_GPIO_0_REGS, 0, 0xffffffff);

    // Match pinmux
    // HWREG(SOC_SYSCFG_0_REGS + SYSCFG0_PINMUX(5)) = 0x81118188;

    // // Point vector table to DDR
    // CP15HighVectorDisable();
    // HWREG(SOC_AINTC_0_REGS + AINTC_VBR) = 0xC0000000;

    // Clean data and instruction caches
    CP15DCacheClean();
    CP15ICacheFlush();

    // Goto kernel entrypoint
    asm volatile(
        "ldr pc, =0xC0000000"
        : : : "memory"
    );
    __builtin_unreachable();

}

static void _drain_write_buffer(void) {
    u32 zero = 0u;

    asm volatile(
        "mcr p15, 0, %0, c7, c10, 4"
        :
        : "r"(zero)
        : "memory"
    );
}
